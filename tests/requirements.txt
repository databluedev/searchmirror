# What the test suite needs ON THE HOST (or CI runner), not in the containers.
#

# backend modules directly to check pure functions -- tests/test_ai_completions.py

# backend/requirements.txt; on the host it has to be declared here.

# add it here or CI fails with ModuleNotFoundError -- which is the point.
pytest
requests
beautifulsoup4
